#version 330 core
out vec4 FragColor;

in vec3 Normal;
in vec3 FragPos;
in vec2 TexCoord;

struct Material
{
	sampler2D diffuseColor;
	sampler2D specularColor;
	float shininess;
};

struct Light
{
	vec3 position;
	vec3 color;
	
	vec3 aIntense;
	vec3 dIntense;
	vec3 sIntense;
};

struct Camera
{
	vec3 position;
	vec3 direction;
};

uniform Material material;
uniform Light light;
uniform Camera camera;

void main()
{
	//	AMBIENT LIGHTING
	vec3 ambient = light.color * light.aIntense * vec3(texture(material.diffuseColor, TexCoord));
	
	//	DIFFUSE LIGHTING
	vec3 normal = normalize(Normal);
	vec3 lightDir = normalize(light.position - FragPos);
	float iDiffuse = max(dot(normal, lightDir), 0.0);
	vec3 diffuse = (light.dIntense * light.color) * (iDiffuse * vec3(texture(material.diffuseColor, TexCoord)));
	
	vec3 result = (ambient + diffuse);
	FragColor = vec4(result, 1.0);
}


/*
	O cálculo de iluminação difusa gera um float que varia de 0 a 1 dependendo da posição da luz, isso porque esse valor é obtido através do cosseno do ângulo entre o vetor de direção da luz, e o vetor normal de cada fragmento, que obtemos através do produto escalar (dot()), dessa forma, se o o ângulo desses vetores forem paralelos, ou seja, 0 graus, o cosseno resulta em 1.0, (assumimos a iluminação máxima, pois o vetor da luz esta alinhado com a direção do fragmento), se o angulo for perpendicular (90 graus), então o cosseno retorna 0.0 (assumindo que a luz não tem contato nenhum com o fragmento). Então conseguimos, a partir dai, multiplicar esse valor pelo vetor de cor do objeto.
*/
